def custom_openapi(app: FastAPI) -> Dict[str, Any]:
    """
    Generate custom OpenAPI schema with enhanced metadata.

    This function is assigned to app.openapi to override
    the default schema generation.

    Args:
        app: FastAPI application instance

    Returns:
        OpenAPI schema dictionary
    """
    if app.openapi_schema:
        return app.openapi_schema

    openapi_schema = get_openapi(
        title=OPENAPI_INFO["title"],
        version=OPENAPI_INFO["version"],
        description=OPENAPI_INFO["description"],
        routes=app.routes,
        tags=TAGS_METADATA,
    )

    # Add contact info
    openapi_schema["info"]["contact"] = OPENAPI_INFO["contact"]

    # Add license info
    openapi_schema["info"]["license"] = OPENAPI_INFO["license_info"]

    # Add external documentation link
    openapi_schema["externalDocs"] = {
        "description": "HX-Infrastructure Documentation",
        "url": "https://github.com/Hana-X-AI/HX-Infrastructure",
    }

    # Add servers information
    openapi_schema["servers"] = [
        {
            "url": "http://hx-lang-server.hx.dev.local:8100",
            "description": "Development server",
        },
        {
            "url": "http://localhost:8100",
            "description": "Local development",
        },
    ]

    # Add security schemes (for documentation, not enforcement)
    openapi_schema["components"]["securitySchemes"] = {
        "sessionId": {
            "type": "apiKey",
            "in": "header",
            "name": "X-Session-ID",
            "description": "Session identifier for request grouping and rate limiting",
        },
        "requestId": {
            "type": "apiKey",
            "in": "header",
            "name": "X-Request-ID",
            "description": "Optional request ID for tracing (auto-generated if not provided)",
        },
        "apiKey": {
            "type": "apiKey",
            "in": "header",
            "name": "X-API-Key",
            "description": "API key authentication (Phase 2 - not currently enforced)",
        },
    }

    # Cache the schema
    app.openapi_schema = openapi_schema

    return app.openapi_schema


def configure_openapi(app: FastAPI) -> None:
    """
    Configure OpenAPI documentation for the application.

    Call this from main.py after creating the FastAPI app.

    Args:
        app: FastAPI application instance
    """
    # Set tags metadata
    app.openapi_tags = TAGS_METADATA

    # Override openapi method for custom schema
    app.openapi = lambda: custom_openapi(app)

```bash
def create_app() -> FastAPI:
    """
    Application factory that creates and configures the FastAPI application.

    Returns:
        FastAPI: Configured application instance.

    This pattern enables:
    - Dependency injection for testing
    - Clean separation of configuration and instantiation
    - Multiple app instances for testing
    """
    settings = get_settings()

    app = FastAPI(
        title=__app_name__,
        description=__description__,
        version=__version__,
        docs_url="/docs" if settings.debug else None,
        redoc_url="/redoc" if settings.debug else None,
        openapi_url="/openapi.json",
        openapi_tags=TAGS_METADATA,
        lifespan=lifespan,
    )

    # Configure CORS middleware
    cors_config = get_cors_config(settings)
    app.add_middleware(CORSMiddleware, **cors_config)

    # Configure security middleware (request ID, rate limiting, security headers)
    configure_middleware(app, settings)

    # Configure OpenAPI documentation
    configure_openapi(app)

    # Register API routers
    _register_routers(app)

    return app


def _register_routers(app: FastAPI) -> None:
    """
    Register all API routers with the application.

    Router structure per API Specification:
    - /api/v1/invoke - Synchronous agent invocation
    - /api/v1/stream - Streaming agent invocation (SSE)
    - /api/v1/sessions - Session management
    - /health - Health check
    - /ready - Readiness probe
    - /metrics - Prometheus metrics
    """
    # API v1 routers (port 8100)
    app.include_router(
        invoke.router,
        prefix="/api/v1",
        tags=["invoke"],
    )
    app.include_router(
        stream.router,
        prefix="/api/v1",
        tags=["stream"],
    )
    app.include_router(
        sessions.router,
        prefix="/api/v1",
        tags=["sessions"],
    )

    # Health/monitoring routers
    app.include_router(
        health.router,
        tags=["health"],
    )


# Create application instance for Uvicorn
app = create_app()


if __name__ == "__main__":
    """
    Development server entry point.

    For production, use systemd to start Uvicorn:
    uvicorn app.main:app --host 0.0.0.0 --port 8100
    """
    import uvicorn

    settings = get_settings()
    uvicorn.run(
        "app.main:app",
        host="0.0.0.0",
        port=settings.service_port,
        reload=settings.debug,
        log_level=settings.log_level.lower(),
    )
EOF
```

## Notes

### Interactive Documentation

- **Swagger UI** (`/docs`): Interactive API explorer with "Try it out" functionality
- **ReDoc** (`/redoc`): Clean, readable documentation format
- **OpenAPI JSON** (`/openapi.json`): Raw schema for code generation

### Debug Mode Control

In production (debug=false):
- `/docs` and `/redoc` are disabled
- `/openapi.json` remains available for automated tooling

### Example Integration

Pydantic models include examples via `json_schema_extra`:
```python
Field(..., json_schema_extra={"example": "value"})
```

These examples appear in the OpenAPI schema and Swagger UI.

### API Client Generation

The OpenAPI schema can be used to generate client libraries:
```bash
# Python client
openapi-generator generate -i http://localhost:8100/openapi.json -g python -o ./client

# TypeScript client
openapi-generator generate -i http://localhost:8100/openapi.json -g typescript-fetch -o ./client
```
